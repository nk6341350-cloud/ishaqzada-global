rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() {
      return request.auth != null;
    }

    function isAdmin() {
      return signedIn() && request.auth.token.email == "nk6341350@gmail.com";
    }

    function sellerApproved(uid) {
      return exists(/databases/$(database)/documents/sellerRequests/$(uid)) &&
             get(/databases/$(database)/documents/sellerRequests/$(uid)).data.status == "approved";
    }

    // Shared marketplace data controlled by Admin.
    match /appData/{docId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // Private seller verification/KYC.
    match /sellerRequests/{uid} {
      allow create: if signedIn() &&
                    request.auth.uid == uid &&
                    request.resource.data.ownerUid == request.auth.uid;

      allow read: if isAdmin() ||
                  (signedIn() && request.auth.uid == uid);

      // Admin can change approval status.
      // Seller can only change their own logo after approval.
      allow update: if isAdmin() || (
        signedIn() &&
        request.auth.uid == uid &&
        resource.data.ownerUid == request.auth.uid &&
        request.resource.data.ownerUid == request.auth.uid &&
        request.resource.data.status == resource.data.status &&
        request.resource.data.diff(resource.data).affectedKeys().hasOnly(['logo','updatedAt'])
      );

      allow delete: if isAdmin();
    }

    // Public shop profiles.
    match /publicSellers/{uid} {
      allow read: if true;
      allow create, update: if isAdmin() || (
        signedIn() &&
        request.auth.uid == uid &&
        request.resource.data.ownerUid == request.auth.uid &&
        request.resource.data.status == "approved" &&
        sellerApproved(uid)
      );
      allow delete: if isAdmin();
    }

    // Public products. Approved sellers can publish their own products.
    match /products/{productId} {
      allow read: if true;

      allow create: if signedIn() &&
                    request.resource.data.ownerUid == request.auth.uid &&
                    request.resource.data.status == "approved" &&
                    sellerApproved(request.auth.uid);

      allow update: if isAdmin() || (
        signedIn() &&
        resource.data.ownerUid == request.auth.uid &&
        request.resource.data.ownerUid == resource.data.ownerUid &&
        sellerApproved(request.auth.uid)
      );

      allow delete: if isAdmin() || (
        signedIn() && resource.data.ownerUid == request.auth.uid
      );
    }
  }
}
